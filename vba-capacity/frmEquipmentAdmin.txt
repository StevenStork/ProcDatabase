Option Explicit

Private mHandlers As Collection
Private mEditingCode As String

Private Sub UserForm_Initialize()
    Me.Caption = "Manage Equipment"
    Me.Width = 440
    Me.Height = 320

    Set mHandlers = CreateControlHandlers()
    BuildControls
    LoadRecordList
    ClearEditor
End Sub

Private Sub BuildControls()
    AddLabel Me, "Equipment", 12, 8, 120
    AddListBox Me, "lstRecords", 12, 24, 170, 200

    AddLabel Me, "Equipment Code", 198, 24, 100
    AddTextBox Me, "txtCode", 198, 40, 210

    AddLabel Me, "Equipment Name", 198, 68, 100
    AddTextBox Me, "txtName", 198, 84, 210

    AddLabel Me, "Equipment Type", 198, 112, 100
    AddTextBox Me, "txtType", 198, 128, 210

    AddCheckBox Me, "chkActive", "Active", 198, 156

    AddLabel Me, "Notes", 198, 180, 100
    AddTextBox Me, "txtNotes", 198, 196, 210, 52

    AddCommandButton Me, "cmdNew", "New", 12, 234, 70
    AddCommandButton Me, "cmdSave", "Save", 88, 234, 70
    AddCommandButton Me, "cmdDelete", "Delete", 164, 234, 70
    AddCommandButton Me, "cmdClose", "Close", 338, 234, 70

    WireControlEvent mHandlers, GetControl(Me, "lstRecords"), Me, "RecordsClick", "Click"
    WireControlEvent mHandlers, GetControl(Me, "cmdNew"), Me, "NewRecord", "Click"
    WireControlEvent mHandlers, GetControl(Me, "cmdSave"), Me, "SaveRecord", "Click"
    WireControlEvent mHandlers, GetControl(Me, "cmdDelete"), Me, "DeleteRecord", "Click"
    WireControlEvent mHandlers, GetControl(Me, "cmdClose"), Me, "CloseForm", "Click"
End Sub

Public Sub HandleFormEvent(ByVal eventName As String)
    Select Case eventName
        Case "RecordsClick": LoadSelectedRecord
        Case "NewRecord": ClearEditor
        Case "SaveRecord": SaveCurrentRecord
        Case "DeleteRecord": DeleteCurrentRecord
        Case "CloseForm": Unload Me
    End Select
End Sub

Private Sub LoadRecordList()
    Dim tbl As ListObject
    Dim items As Variant

    Set tbl = FindTable(EQUIPMENT_TABLE_NAME)
    items = ListAllDisplayItems(tbl, COL_EQUIPMENT_CODE, COL_EQUIPMENT_NAME, False)
    BindListBoxFromArray GetControl(Me, "lstRecords"), items
End Sub

Private Sub LoadSelectedRecord()
    Dim tbl As ListObject
    Dim codeValue As String
    Dim listRowIndex As Long

    codeValue = SelectedListCode(GetControl(Me, "lstRecords"))
    If Len(codeValue) = 0 Then Exit Sub

    Set tbl = FindTable(EQUIPMENT_TABLE_NAME)
    listRowIndex = FindListRowByKey(tbl, COL_EQUIPMENT_CODE, codeValue)
    If listRowIndex = 0 Then Exit Sub

    mEditingCode = codeValue
    GetControl(Me, "txtCode").Text = codeValue
    GetControl(Me, "txtCode").Enabled = False
    GetControl(Me, "txtName").Text = CStr(Nz(GetCellValueByListRow(tbl, listRowIndex, COL_EQUIPMENT_NAME)))
    GetControl(Me, "txtType").Text = CStr(Nz(GetCellValueByListRow(tbl, listRowIndex, COL_EQUIPMENT_TYPE)))
    GetControl(Me, "chkActive").Value = IsActiveFlag(GetCellValueByListRow(tbl, listRowIndex, COL_ACTIVE))
    GetControl(Me, "txtNotes").Text = CStr(Nz(GetCellValueByListRow(tbl, listRowIndex, COL_NOTES)))
End Sub

Private Sub ClearEditor()
    mEditingCode = vbNullString
    GetControl(Me, "txtCode").Enabled = True
    GetControl(Me, "txtCode").Text = vbNullString
    GetControl(Me, "txtName").Text = vbNullString
    GetControl(Me, "txtType").Text = vbNullString
    GetControl(Me, "chkActive").Value = True
    GetControl(Me, "txtNotes").Text = vbNullString
    GetControl(Me, "lstRecords").ListIndex = -1
End Sub

Private Sub SaveCurrentRecord()
    Dim tbl As ListObject
    Dim fieldValues As Object
    Dim codeValue As String
    Dim excludeRow As Long

    codeValue = NormalizeCode(GetControl(Me, "txtCode").Text)
    If Not ValidateRequiredCode(codeValue, "Equipment Code") Then Exit Sub

    Set tbl = FindTable(EQUIPMENT_TABLE_NAME)
    If tbl Is Nothing Then
        MsgBox "Equipment table was not found. Run BootstrapCapacityTables first.", vbExclamation
        Exit Sub
    End If

    excludeRow = 0
    If Len(mEditingCode) > 0 Then
        excludeRow = FindListRowByKey(tbl, COL_EQUIPMENT_CODE, mEditingCode)
    End If

    If Not ValidateUniqueKey(tbl, COL_EQUIPMENT_CODE, codeValue, excludeRow) Then Exit Sub

    Set fieldValues = NewFieldValuesDictionary()
    fieldValues(COL_EQUIPMENT_CODE) = codeValue
    fieldValues(COL_EQUIPMENT_NAME) = Trim$(GetControl(Me, "txtName").Text)
    fieldValues(COL_EQUIPMENT_TYPE) = Trim$(GetControl(Me, "txtType").Text)
    fieldValues(COL_ACTIVE) = ActiveFlagToCellValue(CBool(GetControl(Me, "chkActive").Value))
    fieldValues(COL_NOTES) = Trim$(GetControl(Me, "txtNotes").Text)

    On Error GoTo CleanUp
    OptimizeExcel True
    UpsertRow tbl, COL_EQUIPMENT_CODE, codeValue, fieldValues
    mEditingCode = codeValue
    GetControl(Me, "txtCode").Enabled = False
    LoadRecordList
    SelectListItemByCode GetControl(Me, "lstRecords"), codeValue

CleanUp:
    OptimizeExcel False
End Sub

Private Sub DeleteCurrentRecord()
    Dim tbl As ListObject
    Dim codeValue As String

    codeValue = NormalizeCode(GetControl(Me, "txtCode").Text)
    If Len(codeValue) = 0 Then
        MsgBox "Select equipment to delete.", vbExclamation
        Exit Sub
    End If

    If MsgBox("Delete equipment '" & codeValue & "'?", vbQuestion + vbYesNo) <> vbYes Then Exit Sub

    If EquipmentHasAssignments(codeValue) Then
        MsgBox "Remove factory and process assignments before deleting this equipment.", vbExclamation
        Exit Sub
    End If

    Set tbl = FindTable(EQUIPMENT_TABLE_NAME)
    On Error GoTo CleanUp
    OptimizeExcel True
    DeleteRowByKey tbl, COL_EQUIPMENT_CODE, codeValue
    ClearEditor
    LoadRecordList

CleanUp:
    OptimizeExcel False
End Sub

Private Function EquipmentHasAssignments(ByVal equipmentCode As String) As Boolean
    Dim factoryEquipmentTbl As ListObject
    Dim equipmentProcessTbl As ListObject

    Set factoryEquipmentTbl = FindTable(FACTORY_EQUIPMENT_TABLE_NAME)
    Set equipmentProcessTbl = FindTable(EQUIPMENT_PROCESSES_TABLE_NAME)

    EquipmentHasAssignments = JunctionTableHasEquipment(factoryEquipmentTbl, equipmentCode) _
        Or JunctionTableHasEquipment(equipmentProcessTbl, equipmentCode)
End Function

Private Function JunctionTableHasEquipment(ByVal junctionTbl As ListObject, ByVal equipmentCode As String) As Boolean
    Dim filterValues As Variant
    Dim rowIndex As Long

    JunctionTableHasEquipment = False
    If junctionTbl Is Nothing Then Exit Function
    If junctionTbl.DataBodyRange Is Nothing Then Exit Function

    filterValues = junctionTbl.ListColumns(COL_EQUIPMENT_CODE).DataBodyRange.Value2
    If Not IsArray(filterValues) Then
        JunctionTableHasEquipment = ValuesMatchCode(filterValues, equipmentCode)
        Exit Function
    End If

    For rowIndex = 1 To UBound(filterValues, 1)
        If ValuesMatchCode(filterValues(rowIndex, 1), equipmentCode) Then
            JunctionTableHasEquipment = True
            Exit Function
        End If
    Next rowIndex
End Function
